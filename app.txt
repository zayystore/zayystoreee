<script>
// GANTI nomor berikut dengan nomor WhatsApp penjual.
// Gunakan kode negara 62, tanpa tanda + atau angka 0 di depan.
const WHATSAPP_NUMBER = "6281234567890";

const productsData = [
  {id:1, name:"Sneakers Urban", category:"Fashion",
   price:249000, image:"https://images.unsplash.com/photo-1542291026-7eec264c27ff?auto=format&fit=crop&w=600&q=80"},

  {id:3, name:"Headphone Wireless", category:"Elektronik",
   price:329000, image:"https://images.unsplash.com/photo-1505740420928-5e560c06d30e?auto=format&fit=crop&w=600&q=80"},
  {id:4, name:"Tas Minimalis", category:"Fashion",
   price:199000, image:"https://images.unsplash.com/photo-1548036328-c9fa89d128fa?auto=format&fit=crop&w=600&q=80"},
  {id:5, name:"Kamera Instan", category:"Elektronik",
   price:450000, image:"https://images.unsplash.com/photo-1516035069371-29a1b244cc32?auto=format&fit=crop&w=600&q=80"},
  {id:6, name:"Kacamata Modern", category:"Aksesori",
   price:99000, image:"https://images.unsplash.com/photo-1511499767150-a48a237f0083?auto=format&fit=crop&w=600&q=80"},
  {id:7, name:"Kursi Minimalis", category:"Rumah",
   price:375000, image:"https://images.unsplash.com/photo-1503602642458-232111445657?auto=format&fit=crop&w=600&q=80"},
  {id:8, name:"Lampu Dekorasi", category:"Rumah",
   price:125000, image:"https://images.unsplash.com/photo-1507473885765-e6ed057f782c?auto=format&fit=crop&w=600&q=80"}
];

const rupiah = n => new Intl.NumberFormat("id-ID", {
  style:"currency", currency:"IDR", maximumFractionDigits:0
}).format(n);

let cart = {};
let activeCategory = "Semua";
let searchTerm = "";

try {
  cart = JSON.parse(localStorage.getItem("zayyCart") || "{}") || {};
} catch {
  cart = {};
}

function saveCart() {
  try {
    localStorage.setItem("zayyCart", JSON.stringify(cart));
  } catch (e) {
    console.warn("Keranjang tidak dapat disimpan di browser.");
  }
}

function renderCategories() {
  const categories = ["Semua", ...new Set(productsData.map(p=>p.category))];
  document.getElementById("categories").innerHTML = categories.map(c=>
    `<button class="chip ${c===activeCategory?"active":""}"
      onclick="setCategory('${c}')">${c}</button>`
  ).join("");
}

function setCategory(category) {
  activeCategory = category;
  renderCategories();
  renderProducts();
}

function renderProducts() {
  const filtered = productsData.filter(p =>
    (activeCategory==="Semua" || p.category===activeCategory) &&
    p.name.toLowerCase().includes(searchTerm.toLowerCase())
  );
  document.getElementById("productCount").textContent =
    filtered.length + " produk";
  document.getElementById("products").innerHTML = filtered.length
    ? filtered.map(p=>`
      <article class="product">
        <img src="${p.image}" alt="${p.name}" loading="lazy"
          onerror="this.style.display='none'">
        <span class="tag">${p.category}</span>
        <h3>${p.name}</h3>
        <div class="price">${rupiah(p.price)}</div>
        <div class="product-bottom">
          <span class="muted">Produk pilihan</span>
          <button class="add" onclick="addToCart(${p.id})">+ Tambah</button>
        </div>
      </article>
    `).join("")
    : '<p class="empty">Produk tidak ditemukan.</p>';
}

function addToCart(id) {
  cart[id] = (cart[id] || 0) + 1;
  saveCart();
  renderCart();
}

function changeQty(id, delta) {
  cart[id] = (cart[id] || 0) + delta;
  if (cart[id] <= 0) delete cart[id];
  saveCart();
  renderCart();
}

function cartDetails() {
  return productsData
    .filter(p => cart[p.id] > 0)
    .map(p => ({...p, qty:cart[p.id]}));
}

function renderCart() {
  const items = cartDetails();
  const count = items.reduce((sum,p)=>sum+p.qty,0);
  const total = items.reduce((sum,p)=>sum+p.price*p.qty,0);
  document.getElementById("cartCount").textContent = count;
  document.getElementById("total").textContent = rupiah(total);

  document.getElementById("cartItems").innerHTML = items.length
    ? items.map(p=>`
      <div class="cart-row">
        <div>
          <h4>${p.name}</h4>
          <span class="muted">${rupiah(p.price)} / produk</span>
          <div style="margin-top:7px;font-weight:700">${rupiah(p.price*p.qty)}</div>
        </div>
        <div class="cart-actions">
          <button class="qty" onclick="changeQty(${p.id},-1)">−</button>
          <span>${p.qty}</span>
          <button class="qty" onclick="changeQty(${p.id},1)">+</button>
          <button class="remove" onclick="removeItem(${p.id})">Hapus</button>
        </div>
      </div>
    `).join("")
    : '<p class="empty">Keranjang masih kosong. Yuk, pilih produk favoritmu!</p>';
}

function removeItem(id) {
  delete cart[id];
  saveCart();
  renderCart();
}

function goCart() {
  document.getElementById("cart").scrollIntoView({behavior:"smooth"});
}

function goCheckout() {
  if (!cartDetails().length) {
    alert("Keranjang masih kosong. Tambahkan produk terlebih dahulu.");
    return;
  }
  document.getElementById("checkout").scrollIntoView({behavior:"smooth"});
}

document.getElementById("search").addEventListener("input", e=>{
  searchTerm = e.target.value;
  renderProducts();
});

document.getElementById("checkoutForm").addEventListener("submit", e=>{
  e.preventDefault();
  const items = cartDetails();
  if (!items.length) {
    alert("Keranjang masih kosong.");
    goCart();
    return;
  }

  const name = document.getElementById("name").value.trim();
  const phone = document.getElementById("phone").value.trim();
  const address = document.getElementById("address").value.trim();
  if (!name || !phone || !address) {
    alert("Lengkapi nama, nomor WhatsApp, dan alamat.");
    return;
  }

  const total = items.reduce((sum,p)=>sum+p.price*p.qty,0);
  const order = items.map(p =>
    `- ${p.name} x${p.qty} = ${rupiah(p.price*p.qty)}`
  ).join("\n");

  const message = [
    "HALO ZAYY STORE, SAYA INGIN MEMESAN:",
    "",
    order,
    "",
    `Total produk: ${rupiah(total)}`,
    "",
    `Nama: ${name}`,
    `WhatsApp pembeli: ${phone}`,
    `Alamat: ${address}`,
    `Pengiriman: ${document.getElementById("shipping").value}`,
    `Pembayaran: ${document.getElementById("payment").value}`,
    `Catatan: ${document.getElementById("notes").value.trim() || "-"}`,
    "",
    "Mohon konfirmasi total akhir dan ketersediaan produk. Terima kasih."
  ].join("\n");

  if (!/^\d{9,15}$/.test(WHATSAPP_NUMBER)) {
    alert("Pemilik toko harus mengatur nomor WhatsApp di kode website.");
    return;
  }

  window.open(
    "https://wa.me/" + WHATSAPP_NUMBER + "?text=" + encodeURIComponent(message),
    "_blank"
  );
});

document.getElementById("year").textContent = new Date().getFullYear();

renderCategories();
renderProducts();
renderCart();
</script>
